fanout_["U1/A"] = {{"U1/Y", 0.3f}, ...} → 从 U1/A 出发的边
fanin_["U2/A"] = {{"U1/Y", 0.0f}, ...} → 到达 U2/A 的边
你用 pin_map_ 来完成步骤 1（名字解析），然后在 finalize() 里拿解析好的指针去填 fanout_ 和 fanin_（步骤 2），邻接表就建好了。

  class TimingEngine {
  public:
      // 前向传播：计算所有 pin 的 arrival_time
      void propagate_arrival_time(TimingGraph& tg);

      // 反向传播：计算所有 pin 的 req_time
      void propagate_required_time(TimingGraph& tg, float clock_period);

      // 计算单个 pin 的 slack = RAT - AT
      static float slack(const Pin& pin);
  };

  提示：
  - slack 是 static 的，因为它不依赖 TimingEngine 对象的任何状态，只操作传入的 Pin
  - propagate_* 接收 TimingGraph&，通过其 fanout()/fanin()/lookup_pin() 接口操作

      // 1. 重置所有 arrival_time
      for (const auto& cell : tg.cells())
          for (const auto& pin : cell.Pins)
              pin.arrival_time.reset();

      // 2. 找到 startpoints 并 seed

          for (const auto& pin : cell.Pins) {

              if (tg.fanin(qn).empty()) {          // ← 没有扇入 => startpoint
